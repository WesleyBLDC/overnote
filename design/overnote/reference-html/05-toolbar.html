const fmt = (n: number) =>
  n < 10_000 ? n.toLocaleString()
  : n < 100_000 ? (n/1000).toFixed(1)
      .replace(/\.0$/, '') + 'k'
  : n < 1_000_000 ? Math.round(n/1000) + 'k'
  : (n/1e6).toFixed(1).replace(/\.0$/, '') + 'M';
